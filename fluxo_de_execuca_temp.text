# === 0. Setup do ambiente (uma vez) ===
python3 -m venv .venv
source .venv/bin/activate
pip install -r requirements.txt
python -m ipykernel install --user --name ibov_pipeline_env --display-name "Python (IBOV Pipeline)"

# === 1. Quality gate ANTES de qualquer treino ===
pytest                                           # 48 testes, ≥60% coverage

# === 2. Download ===
python -m src.ibov_pipeline.make_dataset

# === 3. EDA (independente do pipeline de produção) ===
jupyter nbconvert --to notebook --execute \
  --ExecutePreprocessor.kernel_name="ibov_pipeline_env" \
  --inplace notebooks/01_eda_ibov.ipynb

# === 4. Pipeline de dados + treino ===
# Opção A — comando único via DVC (recomendado: re-executa só o que mudou):
dvc init                                          # uma vez no setup do repo
dvc repro                                         # roda o DAG completo

# Opção B — execução manual stage a stage:
python -m src.ibov_pipeline.feature_engineering
python -m src.ibov_pipeline.train_baseline
python -m src.ibov_pipeline.train_lstm

# === 5. Relatório analítico (lê do MLflow) ===
jupyter nbconvert --to notebook --execute \
  --ExecutePreprocessor.kernel_name="ibov_pipeline_env" \
  --inplace notebooks/02_lstm_training_report.ipynb

# === 6. Promoção ao Model Registry ===
python -m src.ibov_pipeline.register_model

# === 7. Inspeção dos logs persistidos ===
ls -lt logs/ | head        # arquivos por execução com timestamp
